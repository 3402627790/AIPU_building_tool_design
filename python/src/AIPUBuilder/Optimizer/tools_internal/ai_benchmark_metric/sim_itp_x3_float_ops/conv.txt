from ..engine import register_gt, register_gen_weight, _GEN_WEIGHTS
from AIPUBuilder.core import OpType
from AIPUBuilder.logger import INFO, ERROR
import torch
from .gt_util.aiff import _aiff_compression_bias_combin, _aiff_merge_shifts
from .gt_util.layout_transfer import layout_convert
from .gt_util.aiff import _aiif_ahead_shift_biases, _aiif_ahead_shift_biases_for_merge
from .gt_util.aiff import _aiff_inner_activation, _aiff_final_output
from .gt_util.aiff import clip_to_bit, pad_input
from AIPUBuilder.core import Target
from .gt_util import check_in_datalayout, check_out_datalayout
from .gt_util.quant_utils import (fp_activation,
                                  linear_dequantize,
                                  to_fp24,
                                  clamp,
                                  linear_quantize_clip)
from .dwconv import depthwiseconv
import numpy as np
import time


@register_gt(OpType.Convolution)
def convolution(self, *args, **kwargs):
    if self.inputs[0].dtype.is_float:
        return fp_conv(self)

    if 'is_dwconv' in self.attrs and self.attrs['is_dwconv']:
        x = depthwiseconv(self, *args, **kwargs)
        return x

    self.sgattrs = kwargs['sgattrs'] if 'sgattrs' in kwargs else None
    if self.sgattrs is None:
        self.sgattrs = dict()

    fuse = True
    if not ('fuse' in kwargs and kwargs['fuse'] and
            (OpType.Eltwise in [c.type for c in self.children] or
             OpType.BatchNorm in [c.type for c in self.children] or
             OpType.Activation in [c.type for c in self.children])):
        fuse = False

    inp = self.inputs[0].betensor.long()
    inp = check_in_datalayout(self, [inp], ['NHWC'])[0]

    with_winograd = self.params['with_winograd'] if 'with_winograd' in self.params else False

    stride = (self.params["stride_y"], self.params["stride_x"])
    dilation = (self.params['dilation_y'], self.params['dilation_x'])
    padding = (self.params['pad_left'], self.params['pad_right'],
               self.params['pad_top'], self.params['pad_bottom'])

    bias_type = self.constants['biases'].betensor_type
    time0 = time.time()
    bias = _aiff_compression_bias_combin(self.constants['biases'].betensor.type(bias_type))
    requant_scale = 1
    requant_shift = 0
    if 'scale_value' in self.params:
        requant_scale = self.params['scale_value']
    elif "scale" in self.constants:
        requant_scale = self.constants["scale"].betensor
    if 'shift_value' in self.params:
        requant_shift = self.params['shift_value']
    elif "shift" in self.constants:
        requant_shift = self.constants["shift"].betensor
    if isinstance(requant_scale, torch.Tensor) and isinstance(requant_shift, torch.Tensor):
        requant_scale, requant_shift = _aiff_merge_shifts(requant_scale.long(), requant_shift.long())
    time1 = time.time()
    # print('time bias_compress: ', time1 - time0)
    weights = self.constants['weights']
    group = self.params['group']
    if 'pads_mode' in self.params:  # for simulator
        inp = pad_input(self, inp)
    inp = layout_convert(inp, _from='NHWC', to='NCHW')
    if 'pads_mode' not in self.params:
        pad_val = -self.inputs[0].quantization.offset
        inp = torch.nn.functional.pad(inp, padding, value=pad_val)
    weights = layout_convert(weights.betensor.long(), _from='NHWC', to='NCHW')
    time2 = time.time()
    # print('time layout_convert: ', time2 - time1)

    split_oc = None
    if 'split_oc' in self.sgattrs:
        split_oc = self.sgattrs['split_oc']
    if split_oc is None or len(split_oc) == 0:
        split_oc = [weights.shape[0]]
    wsoes = list(torch.split(weights, split_oc, dim=0))
    bsoes = list(torch.split(bias, split_oc, dim=0))

    x0 = [None] * len(split_oc)
    x1 = [None] * len(split_oc)
    for i, oc in enumerate(split_oc):
        rounding_when_16bit_merge = False
        if 'rounding_when_16bit_merge' in self.sgattrs \
                and self.sgattrs['rounding_when_16bit_merge'][i] \
                and group == 1 \
                and not with_winograd:
            # TODO: group conv
            # TODO: winograd
            rounding_when_16bit_merge = True
            self.add_bias_after_merge = self.sgattrs['add_bias_after_merge'] if 'add_bias_after_merge' in self.sgattrs else False
            merge_mtp0_c = self.sgattrs['merge_mtp0_c']
            merge_mtp1_c = self.sgattrs['merge_mtp1_c']

            inps = torch.split(inp, (merge_mtp0_c, merge_mtp1_c), dim=1)
            wsies = torch.split(wsoes[i], (merge_mtp0_c, merge_mtp1_c), dim=1)

            inps = [_inp.float().to('cuda') if torch.cuda.is_available() else _inp.long().to('cpu') for _inp in inps]
            wsies = [wsi.float().to('cuda') if torch.cuda.is_available() else wsi.long().to('cpu') for wsi in wsies]

            x0[i] = torch.nn.functional.conv2d(inps[0], wsies[0],
                                               bias=None,
                                               stride=stride,
                                               padding=0,
                                               groups=group,
                                               dilation=dilation)
            x0[i] = layout_convert(x0[i], _from='NCHW', to='NHWC')
            x1[i] = torch.nn.functional.conv2d(inps[1], wsies[1],
                                               bias=None,
                                               stride=stride,
                                               padding=0,
                                               groups=group,
                                               dilation=dilation)
            x1[i] = layout_convert(x1[i], _from='NCHW', to='NHWC')

            # ir_out_shape = x0[i].shape  # x0[i] and x1[i] shape are same
        else:
            # gpu do not support long/double in torch.conv_transpose2d
            inp = inp.float().to('cuda') if torch.cuda.is_available() else inp.long().to('cpu')
            wsoes[i] = wsoes[i].float().to('cuda') if torch.cuda.is_available() else wsoes[i].long().to('cpu')

            # print('device inp2: ', inp.device)
            # print('device weights2: ', wsoes[i].device)

            time11 = time.time()
            x0[i] = torch.nn.functional.conv2d(inp, wsoes[i],
                                               bias=None,
                                               stride=stride,
                                               padding=0,
                                               dilation=dilation,
                                               groups=group)
            # print('device: ', x0[i].device)
            time3 = time.time()
            # print('time conv2d: ', time3 - time11)
            x0[i] = layout_convert(x0[i], _from='NCHW', to='NHWC')
            time4 = time.time()
            # print('time layout_convert: ', time4 - time3)
            # ir_out_shape = x0[i].shape

        if rounding_when_16bit_merge:
            # print("%s will use merge method" % self.name)
            x0[i], remain_shift = _aiif_ahead_shift_biases_for_merge(
                self, x0[i], x1[i], origin_shift=int(requant_shift), biases=bsoes[i])
        else:
            x0[i], remain_shift = _aiif_ahead_shift_biases(self, x0[i], origin_shift=int(requant_shift), biases=bsoes[i])
            x0[i] = clip_to_bit(x0[i], 33, True)  # for itp 33bit m0_alu
        time5 = time.time()
        # print('time _aiif_ahead_shift_biases: ', time5 - time4)
    if len(x0) != 1:
        x = torch.cat(x0, axis=3)
    else:
        x = x0[0]
    ir_out_shape = x.shape

    if isinstance(requant_scale, list):
        full_scale = []
        for s in requant_scale:
            full_scale += [s] * (ir_out_shape[-1] // len(requant_scale))
        requant_scale = full_scale
    x, final_scale, final_shift = _aiff_inner_activation(self, x, requant_scale, remain_shift)
    time6 = time.time()
    # print('time _aiff_inner_activation: ', time6 - time5)
    # final doquant and -zp
    x = _aiff_final_output(self, x, final_scale, final_shift, fuse=fuse)
    x = clip_to_bit(x, 32, True)
    time7 = time.time()
    # print('time _aiff_final_output: ', time7 - time6)
    o_type = str(self.outputs[0].dtype)
    o_bit = eval(o_type.split('int')[-1])
    sign = 'u' not in o_type
    #uint [0, 2**o_bit-1]
    #int: [-2**(obit-1), 2**(o_bit-1)-1]
    if not fuse:
        x = clip_to_bit(x, o_bit, sign=sign, to_type=o_type)
    # else:
    #     print("%s will fuse" % self.name)
    o_type = str(self.outputs[0].dtype)
    o_bit = eval(o_type.split('int')[-1])
    sign = 'u' not in o_type

    sim_x3_itp = self.attrs.get('sim_x3_itp', '')
    fp_bits = self.attrs.get('sim_x3_itp_fp_bits', 24)
    if 'sim_x3_itp' not in self.attrs:
        print(f"sim_x3_itp not in {self}")
    if 'sim_x3_itp_fp_bits' not in self.attrs:
        print(f"sim_x3_itp_fp_bits not in {self}")
    if sim_x3_itp == 'qdqs':
        from .gt_util.quant_utils import dequantize_with_scale_approx_method, quantize_with_scale_approx_method

        inp_scale_fp16 = torch.tensor(self.inputs[0].quantization.scales, device=inp.device).half()
        weight_scale_fp16 = torch.tensor(self.constants['weights'].quantization.scales, device=inp.device).half()
        out_scale_fp16 = torch.tensor(self.outputs[0].quantization.scales, device=inp.device).half()
        out_zp = torch.tensor(self.outputs[0].quantization.offsets, device=inp.device)

        inp_weight_scale = inp_scale_fp16.float() * weight_scale_fp16.float()

        # inception_v3: 1.295 --> 1.3337
        # inp_weight_scale = to_fp24(inp_scale) * to_fp24(weight_scale)

        # try: scale: --> fp16 --> fp24
        # dequantize using dequantize op from exe
        # quantize using quantize op from exe
        inp_weight_scale = to_fp24(inp_weight_scale, fp_bits)
        out_scale = to_fp24(out_scale_fp16.float(), fp_bits)
        out_zp = to_fp24(out_zp.float(), fp_bits)

        # 1. mtp result should be converted to fp24, not int32 data
        x_fp24 = to_fp24(x.float(), fp_bits)
        # x_fp24 = x
        # 2. then dequantize to fp24 float dtype
        dequantize_x = dequantize_with_scale_approx_method(x_fp24, inp_weight_scale)
        dequantize_x = to_fp24(dequantize_x, fp_bits)

        dequantize_bias = 0
        if 'biases' in self.constants:
            bias_scale_fp16 = torch.tensor(self.constants['biases'].quantization.scales, device=inp.device).half()
            bias_scale = to_fp24(bias_scale_fp16.float(), fp_bits)
            bias_fp24 = to_fp24(bias.float(), fp_bits)
            # bias_fp24 = bias
            # 3. dequantize bias to fp24 float dtype
            dequantize_bias = dequantize_with_scale_approx_method(bias_fp24, bias_scale)
            dequantize_bias = to_fp24(dequantize_bias, fp_bits)

        deq_x_add_bias = to_fp24(dequantize_x + dequantize_bias, fp_bits)
        if not fuse:
            q_x_add_bias = (to_fp24(quantize_with_scale_approx_method(deq_x_add_bias, out_scale), fp_bits) - out_zp).round()
            # q_x_add_bias = (to_fp24(deq_x_add_bias * out_scale, fp_bits) - out_zp).round()
            x = clamp(q_x_add_bias, self.outputs[0].dtype)
        else:
            x = deq_x_add_bias
            self.outputs[0]._attrs['fused'] = True
    elif sim_x3_itp == 'qdq':
        inp_scale = torch.tensor(self.inputs[0].quantization.scales, device=inp.device)
        weight_scale = torch.tensor(self.constants['weights'].quantization.scales, device=inp.device)
        out_scale = torch.tensor(self.outputs[0].quantization.scales, device=inp.device)
        out_zp = torch.tensor(self.outputs[0].quantization.offsets, device=inp.device)

        inp_weight_scale = inp_scale * weight_scale
        inp_weight_scale = to_fp24(1. / inp_weight_scale, fp_bits)
        out_scale = to_fp24(out_scale, fp_bits)
        out_zp = to_fp24(out_zp.float(), fp_bits)

        # 1. mtp result should be converted to fp24, not int32 data
        x_fp24 = to_fp24(x.float(), fp_bits)
        # x_fp24 = x
        # 2. then dequantize to fp24 float dtype
        dequantize_x = to_fp24((x_fp24 + torch.zeros_like(inp_weight_scale)) * inp_weight_scale, fp_bits)

        dequantize_bias = 0
        if 'biases' in self.constants:
            bias_scale = torch.tensor(self.constants['biases'].quantization.scales, device=inp.device)
            bias_scale = to_fp24(1. / bias_scale, fp_bits)
            bias_fp24 = to_fp24(bias.float(), fp_bits)
            # 3. dequantize bias to fp24 float dtype
            dequantize_bias = to_fp24(((bias_fp24 + 0) * bias_scale), fp_bits)

        # import pdb
        # pdb.set_trace()
        deq_x_add_bias = to_fp24(dequantize_x + dequantize_bias, fp_bits)
        with_activation = self.params['with_activation']
        if with_activation in ['NONE', 'RELU', 'CLIP']:
            if with_activation == 'RELU':
                deq_x_add_bias = torch.nn.functional.relu(deq_x_add_bias)
            elif with_activation == 'CLIP':
                clip_max = self.params['clip_max']
                clip_min = self.params['clip_min']
                clip_max_f = to_fp24(clip_max / 256. / out_scale, fp_bits)
                clip_min_f = to_fp24(clip_min / 256. / out_scale, fp_bits)
                print(f"clip_min_f={clip_min_f}, clip_max_f={clip_max_f}")

                deq_x_add_bias = torch.clip(deq_x_add_bias, clip_min_f, clip_max_f)

        else:
            ERROR(f"now not support with_activation = {with_activation} in conv")

        if not fuse:
            q_x_add_bias = (to_fp24(deq_x_add_bias * out_scale, fp_bits) - out_zp).round()
            x = clamp(q_x_add_bias, self.outputs[0].dtype)
        else:
            x = deq_x_add_bias
            self.outputs[0]._attrs['fused'] = True
        # inp_scale = torch.tensor(self.inputs[0].quantization.scales, device=inp.device)
        # weight_scale = torch.tensor(self.constants['weights'].quantization.scales, device=inp.device)

        # inp_weight_scale = inp_scale * weight_scale

        # out_scale = torch.tensor(self.outputs[0].quantization.scales, device=inp.device)
        # out_zp = torch.tensor(self.outputs[0].quantization.offsets, device=inp.device)

        # # inp_weight_scale = scale_to_fpx(1. / inp_weight_scale)
        # inp_weight_scale = to_fp24(inp_weight_scale)
        # out_scale = to_fp24(out_scale)

        # # x_fp24 = x
        # # x_fp24 = to_fp24(x.float())
        # # dequantize_x = to_fp24(x_fp24 / inp_weight_scale)
        # # dequantize_x = to_fp24((x + torch.zeros_like(inp_weight_scale)) / inp_weight_scale)
        # dequantize_x = to_fp24(linear_dequantize(x, inp_weight_scale, torch.zeros_like(inp_weight_scale)))
        # dequantize_bias = 0.
        # if 'biases' in self.constants:
        #     bias_scale = torch.tensor(self.constants['biases'].quantization.scales, device=inp.device)
        #     bias_scale = to_fp24(bias_scale)
        #     # bias_scale = scale_to_fpx(1. / bias_scale)
        #     # assert torch.allclose(bias_scale, inp_weight_scale), f"bias_scale != weight_scale * inp_scale in node {self}"
        #     dequantize_bias = to_fp24(linear_dequantize(bias, bias_scale, 0))

        # deq_x_add_bias = to_fp24(dequantize_x + dequantize_bias)
        # if not fuse:
        #     q_x_add_bias = (deq_x_add_bias * out_scale - out_zp).round()
        #     x = clamp(q_x_add_bias, self.outputs[0].dtype)
        # else:
        #     x = deq_x_add_bias
        #     # self.outputs[0]._attrs['fused'] = True
    elif sim_x3_itp == 'cast':

        x_fp24 = to_fp24(x.float(), fp_bits)
        bias_fp24 = to_fp24(bias.float(), fp_bits)
        # bias_fp24 = bias.float()
        out_zp = torch.tensor(self.outputs[0].quantization.offsets, device=inp.device)
        out_zp_fp24 = to_fp24(out_zp, fp_bits)

        requant_scale = 1
        requant_shift = 0
        if 'scale_value' in self.params:
            requant_scale = self.params['scale_value']
        elif "scale" in self.constants:
            requant_scale = self.constants["scale"].betensor

        if 'shift_value' in self.params:
            requant_shift = self.params['shift_value']
        elif "shift" in self.constants:
            requant_shift = self.constants["shift"].betensor
        if isinstance(requant_scale, torch.Tensor) and isinstance(requant_shift, torch.Tensor):
            requant_scale, requant_shift = _aiff_merge_shifts(requant_scale.long(), requant_shift.long())

        x_add_bias_fp24 = x_fp24 + bias_fp24
        x = to_fp24(x_add_bias_fp24, fp_bits)
        with_activation = self.params['with_activation']
        if with_activation in ['NONE', 'RELU', 'CLIP']:
            if with_activation == 'RELU':
                x = torch.nn.functional.relu(x)
            elif with_activation == 'CLIP':
                clip_max = self.params['clip_max']
                clip_min = self.params['clip_min']
                clip_max_fp24 = to_fp24(torch.tensor(clip_max, device=x_add_bias_fp24.device), fp_bits)
                clip_min_fp24 = to_fp24(torch.tensor(clip_min, device=x_add_bias_fp24.device), fp_bits)

                x = torch.clip(x, clip_min_fp24, clip_max_fp24)
        else:
            ERROR(f"now not support with_activation = {with_activation} in conv")

        # ts = requant_scale * 0.5 ** requant_shift
        # x = torch.round(to_fp24(x * ts) - out_zp, fp_bits)
        x = torch.round(to_fp24(x * requant_scale, fp_bits) * (0.5 ** requant_shift) - out_zp_fp24)
        if not fuse:
            x = clamp(x, self.outputs[0].dtype)
        else:
            x = x
            self.outputs[0]._attrs['fused'] = True


        # remain_shift = requant_shift
        # if rounding_when_16bit_merge:
        #     # print("%s will use merge method" % self.name)
        #     x, remain_shift = _aiif_ahead_shift_biases_for_merge(
        #         self, x0, x1, origin_shift=int(requant_shift), biases=bias)
        # else:
        #     x, remain_shift = _aiif_ahead_shift_biases(self, x, origin_shift=int(requant_shift), biases=bias_fp24)
        #     x = clip_to_bit(x, 33, True)  # for itp 33bit m0_alu
        # x, final_scale, final_shift = _aiff_inner_activation(self, x, requant_scale, remain_shift)
        # x = _aiff_final_output(self, x, final_scale, final_shift, fuse=fuse)
        # x = clip_to_bit(x, 32, True)
        # if not fuse:
        #     x = clip_to_bit(x, o_bit, sign=sign, to_type=o_type)

    else:
        requant_scale = 1
        requant_shift = 0

        if 'scale_value' in self.params:
            requant_scale = self.params['scale_value']
        elif "scale" in self.constants:
            requant_scale = self.constants["scale"].betensor

        if 'shift_value' in self.params:
            requant_shift = self.params['shift_value']
        elif "shift" in self.constants:
            requant_shift = self.constants["shift"].betensor

        if isinstance(requant_scale, torch.Tensor) and isinstance(requant_shift, torch.Tensor):
            requant_scale, requant_shift = _aiff_merge_shifts(requant_scale.long(), requant_shift.long())

        if rounding_when_16bit_merge:
            # print("%s will use merge method" % self.name)
            x, remain_shift = _aiif_ahead_shift_biases_for_merge(
                self, x0, x1, origin_shift=int(requant_shift), biases=bias)
        else:
            x, remain_shift = _aiif_ahead_shift_biases(self, x, origin_shift=int(requant_shift), biases=bias)
            x = clip_to_bit(x, 33, True)  # for itp 33bit m0_alu
        time5 = time.time()
        # print('time _aiif_ahead_shift_biases: ', time5 - time4)

        if isinstance(requant_scale, list):
            full_scale = []
            for s in requant_scale:
                full_scale += [s] * (ir_out_shape[-1] // len(requant_scale))
            requant_scale = full_scale
        x, final_scale, final_shift = _aiff_inner_activation(self, x, requant_scale, remain_shift)
        time6 = time.time()
        # print('time _aiff_inner_activation: ', time6 - time5)
        # final doquant and -zp
        x = _aiff_final_output(self, x, final_scale, final_shift, fuse=fuse)
        x = clip_to_bit(x, 32, True)
        time7 = time.time()
        # print('time _aiff_final_output: ', time7 - time6)
        #uint [0, 2**o_bit-1]
        #int: [-2**(obit-1), 2**(o_bit-1)-1]
        if not fuse:
            x = clip_to_bit(x, o_bit, sign=sign, to_type=o_type)
        # else:
        #     print("%s will fuse" % self.name)
    x = check_out_datalayout(self, [x], ['NHWC'])[0]
    time8 = time.time()
    # print('time check_out_datalayout: ', time8 - time7)
    self.outputs[0].betensor = x
    return x

def gen_random_weight(self):

    in_dtype = str(self.inputs[0].dtype)
    if not in_dtype.startswith('float'):
        self.constants['biases'].betensor = np.random.randint(-1000, 1000, size=self.constants['biases'].shape,
                                                              dtype=str(self.constants['biases'].dtype))
    if 'scale' in self.constants:
        self.constants['scale'].betensor = np.random.randint(1, 150, self.constants['scale'].shape,
                                                             str(self.constants['scale'].dtype))
    if 'shift' in self.constants:
        self.constants['shift'].betensor = np.random.randint(10, 20, self.constants['shift'].shape,

def fp_conv(self):
    stride = (self.params['stride_y'], self.params['stride_x'])
    dilation = (self.params['dilation_y'], self.params['dilation_x'])
    padding = (self.params['pad_left'], self.params['pad_right'],
               self.params['pad_top'], self.params['pad_bottom'])

    inputs = self.inputs[0].betensor.to(torch.float32)
    inputs = check_in_datalayout(self, [inputs], ['NHWC'])[0]
    inputs = layout_convert(inputs, _from='NHWC', to='NCHW').float()
    inputs = torch.nn.functional.pad(inputs, padding, value=0)

    weights = self.constants['weights'].betensor.to(torch.float32) # [oc, kh, kw, ic]
    weights = weights.permute([1, 2, 3, 0]) # [kh, kw, ic, oc]
    # wt dequant
    weights_zp = self.constants['weights'].quantization.offsets # [oc] or [1] or []
    if len(weights_zp) != 0:
        weights = weights + torch.tensor(weights_zp).to(torch.float32)
    weights_scale = self.constants['weights'].quantization.scales # [oc] or [1] or []
    if len(weights_scale) != 0:
        weights = weights / torch.tensor(weights_scale).to(torch.float32)
    weights = weights.permute([3, 0, 1, 2]) # [oc, kh, kw, ic]
    weights = layout_convert(weights, _from='NHWC', to='NCHW').float() # [ic, oc, kh, kw]

    biases = self.constants['biases'].betensor.to(torch.float32)

    output = torch.nn.functional.conv2d(inputs,
                                        weights,
                                        bias=biases,
                                        stride=stride,
                                        padding=0,
                                        groups=self.params['group'],
                                        dilation=dilation)
    output = output.permute([0, 2, 3, 1])
    output = fp_activation(output, self)
    output = output.to(torch.float16)
    output = check_out_datalayout(self, [output], ['NHWC'])[0]
    self.outputs[0].betensor = output
    return output
