@register_gt(OpType.Mul)
def mul(self, *args, **kwargs):
    if 'with_activation' not in self.params:
        self.params['with_activation'] = 'NONE'
    if str(self.inputs[0].dtype) == 'float16':
        output = mul_float(self)
    else:
        self.params['method'] = 'MUL'
        output = eltwise(self, *args, **kwargs)
        self.params.pop('method')
    self.outputs[0].betensor = output
    return output
